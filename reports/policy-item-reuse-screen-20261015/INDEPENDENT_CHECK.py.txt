# ---- independent LRU via plain dict (insertion order) AND reuse-distance definition
CAP = 4096
def lru_dict(stream):
    """stream: list of (call, width, key). returns per-occurrence (hit, kind)."""
    d = {}; out = []
    for c, w, k in stream:
        if k in d:
            pc = d.pop(k); out.append(('within' if pc == c else 'cross'))
        else:
            out.append(None)
            if len(d) == CAP: del d[next(iter(d))]
        d[k] = c
    return out
class BIT:
    def __init__(s, n): s.n = n; s.t = [0] * (n + 1)
    def add(s, i, v):
        i += 1
        while i <= s.n: s.t[i] += v; i += i & -i
    def pre(s, i):
        i += 1; r = 0
        while i > 0: r += s.t[i]; i -= i & -i
        return r
def reuse_distance(stream):
    """hit iff key seen before and #distinct other keys since its last use < CAP."""
    n = len(stream); bit = BIT(n); last = {}; out = []
    for t, (c, w, k) in enumerate(stream):
        if k in last:
            lt, lc = last[k]
            distinct_between = bit.pre(t - 1) - bit.pre(lt)
            out.append(('within' if lc == c else 'cross') if distinct_between < CAP else None)
            bit.add(lt, -1)
        else:
            out.append(None)
        bit.add(t, 1); last[k] = (t, c)
    return out
